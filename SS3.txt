STEP 1.2: Check who has EXECUTE permission on xp_cmdshell
SELECT 
    p.name AS grantee,
    p.type_desc,
    p.is_disabled,
    pp.permission_name,
    pp.state_desc AS permission_state,
    CASE 
        WHEN p.is_disabled = 1 THEN '❌ Account Disabled - Cannot Execute'
        WHEN pp.permission_name IS NULL THEN '⚠️ No Direct Permission - Check Role Membership'
        WHEN pp.permission_name = 'EXECUTE' AND pp.state_desc = 'GRANT' THEN '✅ Can Execute xp_cmdshell'
        WHEN pp.permission_name = 'EXECUTE' AND pp.state_desc = 'DENY' THEN '❌ Explicitly Denied - Cannot Execute'
        ELSE '⚠️ Check Role Membership'
    END AS execution_status
FROM sys.server_principals p
LEFT JOIN sys.server_permissions pp 
    ON p.principal_id = pp.grantee_principal_id 
    AND pp.permission_name = 'EXECUTE'
    AND pp.class = 0
    AND pp.major_id = 0
WHERE p.type IN ('S', 'U', 'G')
AND p.name NOT LIKE '##%'
ORDER BY p.name;

SAFE OUTPUT:
grantee | type_desc | is_disabled | permission_name | permission_state | execution_status
--------|-----------|-------------|-----------------|------------------|------------------
app_user| SQL_LOGIN | 0           | NULL            | NULL             | ⚠️ No Direct Permission - Check Role Membership

VULNERABLE OUTPUT:
grantee | type_desc | is_disabled | permission_name | permission_state | execution_status
--------|-----------|-------------|-----------------|------------------|------------------
app_user| SQL_LOGIN | 0           | EXECUTE         | GRANT            | ✅ Can Execute xp_cmdshell

STEP 1.3: Check who is in sysadmin (can execute xp_cmdshell by default)
SELECT 
    r.name AS role_name,
    m.name AS member_name,
    m.type_desc,
    m.is_disabled,
    '✅ Can execute xp_cmdshell via sysadmin role' AS execution_status
FROM sys.server_role_members srm
JOIN sys.server_principals r ON srm.role_principal_id = r.principal_id
JOIN sys.server_principals m ON srm.member_principal_id = m.principal_id
WHERE r.name = 'sysadmin'
AND m.is_disabled = 0;

SAFE OUTPUT:

STEP 31.1: Check sensitive columns
SELECT 
    SCHEMA_NAME(t.schema_id) AS schema_name,
    t.name AS table_name,
    c.name AS column_name,
    ty.name AS data_type,
    CASE 
        WHEN ty.name IN ('varchar', 'nvarchar', 'text', 'ntext') 
        THEN '✅ PLAINTEXT SENSITIVE DATA'
        WHEN ty.name IN ('varbinary', 'binary') 
        THEN '⚠️ Potentially encrypted - Check'
        ELSE 'Check data type'
    END AS sensitive_status
FROM sys.tables t
JOIN sys.columns c ON t.object_id = c.object_id
JOIN sys.types ty ON c.user_type_id = ty.user_type_id
WHERE c.name LIKE '%password%' OR c.name LIKE '%pass%'
   OR c.name LIKE '%secret%' OR c.name LIKE '%token%'

   OR c.name LIKE '%auth%' OR c.name LIKE '%credential%'
   OR c.name LIKE '%salary%' OR c.name LIKE '%sin%'
