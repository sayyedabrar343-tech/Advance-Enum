COMMAND 1: XP_CMDSHELL - OS COMMAND EXECUTION (WITH ACTUAL TEST - ONLY IF VULNERABLE)
================================================================

-- STEP 1: Check status and permissions (hamesha chalega)
SELECT 
    '=== XP_CMDSHELL STATUS ===' AS section,
    SUSER_SNAME() AS current_user,
    CASE WHEN IS_SRVROLEMEMBER('sysadmin') = 1 THEN 'YES' ELSE 'NO' END AS am_i_sysadmin,
    CASE WHEN HAS_PERMS_BY_NAME('xp_cmdshell', 'OBJECT', 'EXECUTE') = 1 THEN 'YES' ELSE 'NO' END AS do_i_have_permission,
    (SELECT value_in_use FROM sys.configurations WHERE name = 'xp_cmdshell') AS is_enabled,
    CASE 
        WHEN (SELECT value_in_use FROM sys.configurations WHERE name = 'xp_cmdshell') = 1 
             AND (IS_SRVROLEMEMBER('sysadmin') = 1 OR HAS_PERMS_BY_NAME('xp_cmdshell', 'OBJECT', 'EXECUTE') = 1)
        THEN 'VULNERABLE - YOU can execute OS commands - CRITICAL'
        WHEN (SELECT value_in_use FROM sys.configurations WHERE name = 'xp_cmdshell') = 1 
             AND EXISTS (SELECT 1 FROM sys.server_role_members srm JOIN sys.server_principals m ON srm.member_principal_id = m.principal_id WHERE srm.role_principal_id = (SELECT principal_id FROM sys.server_principals WHERE name = 'sysadmin') AND m.name != SUSER_SNAME() AND m.is_disabled = 0)
        THEN 'VULNERABLE EXISTS - Other users can execute - Check below'
        WHEN (SELECT value_in_use FROM sys.configurations WHERE name = 'xp_cmdshell') = 1 
        THEN 'WARNING - xp_cmdshell ENABLED but you cannot execute'
        ELSE 'SAFE - xp_cmdshell DISABLED'
    END AS vulnerability_status
UNION ALL
SELECT 
    '=== OTHER VULNERABLE USERS (SYSADMIN) ===',
    STRING_AGG(m.name, ' | '),
    'These users can execute xp_cmdshell',
    'YES',
    'YES'
FROM sys.server_role_members srm
JOIN sys.server_principals m ON srm.member_principal_id = m.principal_id
WHERE srm.role_principal_id = (SELECT principal_id FROM sys.server_principals WHERE name = 'sysadmin')
AND m.name != SUSER_SNAME() AND m.is_disabled = 0
UNION ALL
SELECT 
    '=== USERS WITH EXPLICIT EXECUTE ===',
    STRING_AGG(p.name, ' | '),
    'These users have direct EXECUTE permission',
    'YES',
    'YES'
FROM sys.server_principals p
JOIN sys.server_permissions sp ON p.principal_id = sp.grantee_principal_id
WHERE sp.permission_name = 'EXECUTE' AND sp.major_id = 0 AND sp.class = 0
AND p.name != SUSER_SNAME() AND p.is_disabled = 0;

-- STEP 2: ACTUAL EXECUTION TEST - ONLY IF CURRENT USER IS VULNERABLE
IF EXISTS (SELECT 1 FROM sys.configurations WHERE name = 'xp_cmdshell' AND value_in_use = 1)
AND (IS_SRVROLEMEMBER('sysadmin') = 1 OR HAS_PERMS_BY_NAME('xp_cmdshell', 'OBJECT', 'EXECUTE') = 1)
BEGIN
    PRINT '========================================';
    PRINT 'CONFIRMED VULNERABLE - ACTUAL EXECUTION TEST:';
    PRINT '========================================';
    EXEC xp_cmdshell 'whoami';
    PRINT '========================================';
    PRINT 'CONFIRMED - You can execute OS commands';
    PRINT '========================================';
END
ELSE IF EXISTS (SELECT 1 FROM sys.configurations WHERE name = 'xp_cmdshell' AND value_in_use = 1)
BEGIN
    PRINT '========================================';
    PRINT 'xp_cmdshell is ENABLED but you do NOT have permission';
    PRINT 'Other users may be vulnerable - Check above list';
    PRINT '========================================';
END
ELSE
BEGIN
    PRINT '========================================';
    PRINT 'SAFE - xp_cmdshell is DISABLED';
    PRINT '========================================';
END

COMMAND 2: SA ACCOUNT - BRUTE FORCE TARGET
================================================================

SELECT 
    '=== SA ACCOUNT STATUS ===' AS section,
    SUSER_SNAME() AS current_user,
    CASE WHEN EXISTS (SELECT 1 FROM sys.sql_logins WHERE name = 'sa' AND is_disabled = 0) THEN 'ENABLED' ELSE 'DISABLED' END AS sa_status,
    CASE 
        WHEN EXISTS (SELECT 1 FROM sys.sql_logins WHERE name = 'sa' AND is_disabled = 0 AND is_policy_checked = 0 AND is_expiration_checked = 0)
        THEN 'VULNERABLE - SA ENABLED with NO policy and NO expiration - CRITICAL - Immediate disable required'
        WHEN EXISTS (SELECT 1 FROM sys.sql_logins WHERE name = 'sa' AND is_disabled = 0 AND is_policy_checked = 0)
        THEN 'VULNERABLE - SA ENABLED with NO password policy - Brute force possible - HIGH'
        WHEN EXISTS (SELECT 1 FROM sys.sql_logins WHERE name = 'sa' AND is_disabled = 0)
        THEN 'WARNING - SA ENABLED with policy - Still a brute force target'
        ELSE 'SAFE - SA DISABLED'
    END AS sa_vulnerability
UNION ALL
SELECT 
    '=== SA DETAILS ===',
    'Disabled: ' + CAST(is_disabled AS VARCHAR) + ' | Policy: ' + CAST(is_policy_checked AS VARCHAR) + ' | Expiration: ' + CAST(is_expiration_checked AS VARCHAR),
    'Check each setting'
FROM sys.sql_logins WHERE name = 'sa';

COMMAND 3: IMPERSONATION - PRIVILEGE ESCALATION
================================================================

SELECT 
    '=== IMPERSONATION STATUS ===' AS section,
    SUSER_SNAME() AS current_user,
    CASE WHEN EXISTS (SELECT 1 FROM sys.server_permissions WHERE permission_name = 'IMPERSONATE' AND grantee_principal_id = USER_ID()) THEN 'YES' ELSE 'NO' END AS can_i_impersonate,
    CASE 
        WHEN EXISTS (SELECT 1 FROM sys.server_permissions WHERE permission_name = 'IMPERSONATE' AND major_id = (SELECT principal_id FROM sys.server_principals WHERE name = 'sa') AND grantee_principal_id = USER_ID())
        THEN 'VULNERABLE - YOU can IMPERSONATE SA - Can become sysadmin - CRITICAL'
        WHEN EXISTS (SELECT 1 FROM sys.server_permissions WHERE permission_name = 'IMPERSONATE' AND grantee_principal_id = USER_ID())
        THEN 'VULNERABLE - YOU have IMPERSONATE permission - Check targets below'
        WHEN EXISTS (SELECT 1 FROM sys.server_permissions WHERE permission_name = 'IMPERSONATE')
        THEN 'VULNERABLE EXISTS - Other users can impersonate - Check below'
        ELSE 'SAFE - No impersonation permissions'
    END AS impersonation_status
UNION ALL
SELECT 
    '=== ALL IMPERSONATION GRANTS ===',
    STRING_AGG(CONCAT(g.name, ' -> ', t.name), ' | '),
    'Grantee -> Target (Who can impersonate whom)'
FROM sys.server_permissions sp
JOIN sys.server_principals g ON sp.grantee_principal_id = g.principal_id
JOIN sys.server_principals t ON sp.major_id = t.principal_id
WHERE sp.permission_name = 'IMPERSONATE';

-- Database-level impersonation check (run in each DB)
SELECT 
    '=== DATABASE-LEVEL IMPERSONATION ===' AS section,
    DB_NAME() AS database_name,
    STRING_AGG(CONCAT(g.name, ' -> ', t.name), ' | ') AS db_impersonation_grants,
    CASE 
        WHEN EXISTS (SELECT 1 FROM sys.database_permissions WHERE permission_name = 'IMPERSONATE' AND grantee_principal_id = USER_ID())
        THEN 'VULNERABLE - You can impersonate in this database'
        ELSE 'Check if any impersonation exists'
    END AS db_impersonation_status
FROM sys.database_permissions sp
JOIN sys.database_principals g ON sp.grantee_principal_id = g.principal_id
JOIN sys.database_principals t ON sp.major_id = t.principal_id
WHERE sp.permission_name = 'IMPERSONATE';

        THEN 'WARNING - TRUSTWORTHY ON exists - Check if you can get db_owner'
        ELSE 'SAFE - No TRUSTWORTHY databases'
    END AS trustworthy_status
UNION ALL
SELECT 
    '=== TRUSTWORTHY DATABASES ===',

    'Database (Owner | You have CONTROL?)'

        THEN 'WARNING - CLR ENABLED with strict security ON'

COMMAND 10: ORPHANED USERS - PERSISTENCE (DATABASE-LEVEL)
================================================================

    '=== ORPHANED USERS STATUS ===' AS section,
    DB_NAME() AS database_name,

    CASE

    DB_NAME(),
    STRING_AGG(dp.name, ' | '),
    'Orphaned database users'

COMMAND 11: SENSITIVE COLUMNS - DATA EXPOSURE (DATABASE-LEVEL)
================================================================

-- Run this in EACH database
SELECT 
    '=== SENSITIVE COLUMNS STATUS ===' AS section,
    DB_NAME() AS database_name,
    SUSER_SNAME() AS current_user,
    CAST((SELECT COUNT(*) FROM sys.tables t JOIN sys.columns c ON t.object_id = c.object_id WHERE c.name LIKE '%password%' OR c.name LIKE '%credit%' OR c.name LIKE '%ssn%' OR c.name LIKE '%apikey%' OR c.name LIKE '%secret%' OR c.name LIKE '%token%') AS VARCHAR) AS sensitive_columns_count,
    CASE 
        WHEN EXISTS (SELECT 1 FROM sys.tables t JOIN sys.columns c ON t.object_id = c.object_id JOIN sys.types ty ON c.user_type_id = ty.user_type_id WHERE (c.name LIKE '%password%' OR c.name LIKE '%credit%' OR c.name LIKE '%ssn%' OR c.name LIKE '%apikey%' OR c.name LIKE '%secret%' OR c.name LIKE '%token%') AND ty.name IN ('varchar', 'nvarchar', 'text', 'ntext'))
        THEN 'VULNERABLE - Plaintext sensitive data columns found in ' + DB_NAME() + ' - Data breach risk - HIGH'
        ELSE 'SAFE - No plaintext sensitive columns in ' + DB_NAME()
    END AS sensitive_status
UNION ALL
SELECT 
    '=== SENSITIVE COLUMNS ===',
    DB_NAME(),
    STRING_AGG(CONCAT(SCHEMA_NAME(t.schema_id), '.', t.name, '.', c.name, ' (', ty.name, ')'), ' | '),
    'Schema.Table.Column (Data Type)'
FROM sys.tables t 
JOIN sys.columns c ON t.object_id = c.object_id 
JOIN sys.types ty ON c.user_type_id = ty.user_type_id 
WHERE (c.name LIKE '%password%' OR c.name LIKE '%credit%' OR c.name LIKE '%ssn%' OR c.name LIKE '%apikey%' OR c.name LIKE '%secret%' OR c.name LIKE '%token%') 
AND ty.name IN ('varchar', 'nvarchar', 'text', 'ntext');

COMMAND 13: SQL AGENT JOBS - COMMAND EXECUTION
================================================================

SELECT 
    '=== SQL AGENT JOBS STATUS ===' AS section,
    SUSER_SNAME() AS current_user,
    CAST((SELECT COUNT(*) FROM msdb.dbo.sysjobs j JOIN msdb.dbo.sysjobsteps js ON j.job_id = js.job_id WHERE j.enabled = 1 AND js.subsystem IN ('CmdExec', 'PowerShell')) AS VARCHAR) AS dangerous_jobs_count,
    CASE 
        WHEN EXISTS (SELECT 1 FROM msdb.dbo.sysjobs j JOIN msdb.dbo.sysjobsteps js ON j.job_id = js.job_id WHERE j.enabled = 1 AND js.subsystem IN ('CmdExec', 'PowerShell') AND (SUSER_SNAME(j.owner_sid) = SUSER_SNAME() OR HAS_PERMS_BY_NAME(j.name, 'OBJECT', 'ALTER') = 1 OR IS_SRVROLEMEMBER('sysadmin') = 1))
        THEN 'VULNERABLE - Jobs with OS commands AND you can modify/execute them - Can execute arbitrary commands - HIGH'
        WHEN EXISTS (SELECT 1 FROM msdb.dbo.sysjobs j JOIN msdb.dbo.sysjobsteps js ON j.job_id = js.job_id WHERE j.enabled = 1 AND js.subsystem IN ('CmdExec', 'PowerShell'))
        THEN 'WARNING - Jobs with OS commands exist - Check if you can modify/execute'
        ELSE 'SAFE - No jobs with OS commands'
    END AS job_status
UNION ALL
SELECT 
    '=== DANGEROUS JOBS ===',
    STRING_AGG(CONCAT(j.name, ' (', js.subsystem, ')'), ' | '),
    'Job Name (Subsystem)'
FROM msdb.dbo.sysjobs j
JOIN msdb.dbo.sysjobsteps js ON j.job_id = js.job_id
WHERE j.enabled = 1 AND js.subsystem IN ('CmdExec', 'PowerShell');

COMMAND 14: SQL AGENT PROXIES - PRIVILEGE ESCALATION
================================================================

    '=== SQL AGENT PROXIES STATUS ===' AS section,

    CASE

COMMAND 15: DATABASE ENCRYPTION (TDE) - DATA AT REST
================================================================

SELECT 
    '=== DATABASE ENCRYPTION STATUS ===' AS section,
    SUSER_SNAME() AS current_user,
    CASE WHEN EXISTS (SELECT 1 FROM sys.dm_database_encryption_keys WHERE encryption_state = 3 AND DB_NAME(database_id) NOT IN ('master', 'msdb', 'tempdb', 'model')) THEN 'ENCRYPTED' ELSE 'NOT_ENCRYPTED' END AS encryption_status,
    CASE 
        WHEN EXISTS (SELECT 1 FROM sys.dm_database_encryption_keys WHERE encryption_state = 1 AND DB_NAME(database_id) NOT IN ('master', 'msdb', 'tempdb', 'model'))
        THEN 'VULNERABLE - Databases are NOT ENCRYPTED - Data at risk - HIGH'
        WHEN EXISTS (SELECT 1 FROM sys.dm_database_encryption_keys WHERE encryption_state = 3)
        THEN 'SAFE - All databases are encrypted'
        ELSE 'WARNING - No encryption keys found'
    END AS encryption_vulnerability
UNION ALL
SELECT 
    '=== DB ENCRYPTION DETAILS ===',
    STRING_AGG(CONCAT(DB_NAME(database_id), ' (', 
        CASE encryption_state 
            WHEN 1 THEN 'UNENCRYPTED' 
            WHEN 3 THEN 'ENCRYPTED' 
            ELSE 'UNKNOWN' 
        END, ', Alg: ', key_algorithm, ')'), ' | '),
    'Database (Status | Algorithm)'
FROM sys.dm_database_encryption_keys;

COMMAND 16: DATABASE MASTER KEY - WEAK ENCRYPTION
================================================================

SELECT 
    '=== DATABASE MASTER KEY STATUS ===' AS section,
    SUSER_SNAME() AS current_user,
    (SELECT key_length FROM sys.symmetric_keys WHERE name = '##MS_DatabaseMasterKey##') AS key_length,
    (SELECT algorithm_desc FROM sys.symmetric_keys WHERE name = '##MS_DatabaseMasterKey##') AS algorithm,
    CASE 
        WHEN (SELECT key_length FROM sys.symmetric_keys WHERE name = '##MS_DatabaseMasterKey##') >= 256 AND (SELECT algorithm_desc FROM sys.symmetric_keys WHERE name = '##MS_DatabaseMasterKey##') LIKE '%AES%'
        THEN 'SAFE - Strong encryption (AES-256)'
        WHEN (SELECT algorithm_desc FROM sys.symmetric_keys WHERE name = '##MS_DatabaseMasterKey##') LIKE '%DES%' OR (SELECT algorithm_desc FROM sys.symmetric_keys WHERE name = '##MS_DatabaseMasterKey##') LIKE '%TRIPLE%'
        THEN 'VULNERABLE - Weak encryption (DES/Triple DES) - Upgrade required'
        ELSE 'WARNING - Check encryption'
    END AS master_key_status;
