    SUSER_SNAME() AS current_login,
    CASE 
        WHEN IS_SRVROLEMEMBER('sysadmin') = 1 THEN 'YES' 
        ELSE 'NO' 
    END AS am_i_sysadmin,
    CASE 
        WHEN HAS_PERMS_BY_NAME('xp_cmdshell', 'OBJECT', 'EXECUTE') = 1 THEN 'YES' 
        ELSE 'NO' 
    END AS do_i_have_permission,
    (SELECT value_in_use FROM sys.configurations WHERE name = 'xp_cmdshell') AS is_enabled,
    CASE 
        WHEN (SELECT value_in_use FROM sys.configurations WHERE name = 'xp_cmdshell') = 1 
             AND (IS_SRVROLEMEMBER('sysadmin') = 1 OR HAS_PERMS_BY_NAME('xp_cmdshell', 'OBJECT', 'EXECUTE') = 1)
        THEN 'VULNERABLE - YOU can execute OS commands - CRITICAL'
        WHEN (SELECT value_in_use FROM sys.configurations WHERE name = 'xp_cmdshell') = 1 
             AND EXISTS (SELECT 1 FROM sys.server_role_members srm JOIN sys.server_principals m ON srm.member_principal_id = m.principal_id WHERE srm.role_principal_id = (SELECT principal_id FROM sys.server_principals WHERE name = 'sysadmin') AND m.name != SUSER_SNAME() AND m.is_disabled = 0)
        THEN 'VULNERABLE EXISTS - Other users can execute'
        WHEN (SELECT value_in_use FROM sys.configurations WHERE name = 'xp_cmdshell') = 1 
        THEN 'WARNING - xp_cmdshell ENABLED but you cannot execute'
        ELSE 'SAFE - xp_cmdshell DISABLED'
    END AS vulnerability_status;

IF EXISTS (SELECT 1 FROM sys.configurations WHERE name = 'xp_cmdshell' AND value_in_use = 1)
AND (IS_SRVROLEMEMBER('sysadmin') = 1 OR HAS_PERMS_BY_NAME('xp_cmdshell', 'OBJECT', 'EXECUTE') = 1)

END

BEGIN

        WHEN EXISTS (SELECT 1 FROM sys.sql_logins WHERE name = 'sa' AND is_disabled = 1)

        ELSE 'UNKNOWN - SA not found'

COMMAND 3: SERVER IMPERSONATION
================================================================

SELECT 
    '=== SERVER IMPERSONATION STATUS ===' AS section,
    SUSER_SNAME() AS current_login,
    CASE 
        WHEN EXISTS (SELECT 1 FROM sys.server_permissions WHERE permission_name = 'IMPERSONATE' AND grantee_principal_id = USER_ID()) 
        THEN 'YES' 
        ELSE 'NO' 
    END AS can_impersonate,
    CASE 
        WHEN EXISTS (SELECT 1 FROM sys.server_permissions WHERE permission_name = 'IMPERSONATE' AND major_id = (SELECT principal_id FROM sys.server_principals WHERE name = 'sa') AND grantee_principal_id = USER_ID())
        THEN 'VULNERABLE - YOU can IMPERSONATE SA - CRITICAL'
        WHEN EXISTS (SELECT 1 FROM sys.server_permissions WHERE permission_name = 'IMPERSONATE' AND grantee_principal_id = USER_ID())
        THEN 'VULNERABLE - YOU have IMPERSONATE permission'
        WHEN EXISTS (SELECT 1 FROM sys.server_permissions WHERE permission_name = 'IMPERSONATE')
        THEN 'VULNERABLE EXISTS - Other users can impersonate'
        ELSE 'SAFE - No impersonation permissions'
    END AS impersonation_status;

SELECT 
    '=== IMPERSONATION GRANTS ===' AS section,
    g.name AS grantee,
    t.name AS target_principal
FROM sys.server_permissions sp
JOIN sys.server_principals g ON sp.grantee_principal_id = g.principal_id
JOIN sys.server_principals t ON sp.major_id = t.principal_id
WHERE sp.permission_name = 'IMPERSONATE';

COMMAND 22: SENSITIVE COLUMNS - DATA EXPOSURE
================================================================

SELECT 
    '=== SENSITIVE COLUMNS ===' AS section,
    DB_NAME() AS database_name,
    SCHEMA_NAME(t.schema_id) AS schema_name,
    t.name AS table_name,
    c.name AS column_name,
    ty.name AS data_type,
    CASE 
        WHEN ty.name IN ('varchar', 'nvarchar', 'text', 'ntext')
        THEN 'VULNERABLE - Plaintext sensitive data - Data breach risk - HIGH'
        ELSE 'Check - May be encrypted'
    END AS sensitive_status
FROM sys.tables t
JOIN sys.columns c ON t.object_id = c.object_id
JOIN sys.types ty ON c.user_type_id = ty.user_type_id
WHERE c.name LIKE '%password%' OR c.name LIKE '%pass%'
   OR c.name LIKE '%credit%' OR c.name LIKE '%card%'
   OR c.name LIKE '%ssn%' OR c.name LIKE '%social%'
   OR c.name LIKE '%apikey%' OR c.name LIKE '%api_key%'
   OR c.name LIKE '%secret%' OR c.name LIKE '%token%'
   OR c.name LIKE '%auth%' OR c.name LIKE '%credential%'
ORDER BY schema_name, table_name, column_name;
